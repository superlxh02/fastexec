cmake_minimum_required(VERSION 3.20)
project(fastexec LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(example examples/example.cpp)
target_include_directories(example PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
set_target_properties(example PROPERTIES CXX_STANDARD 23) # 示例里的 fastlog 依赖 C++23
add_executable(future_test tests/future_test.cpp)
target_include_directories(future_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
enable_testing()
add_test(NAME future_test COMMAND future_test)
add_executable(future_bench benchmarks/future_bench.cpp)
target_include_directories(future_bench PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
